\section{Methodology: Adaptive SGDE}
\label{sec:method}

\subsection{Problem Formulation and Overview}
\label{sec:overview}

Given an untrimmed video $\mathcal{V} = \{f_t\}_{t=1}^{T}$ of duration $T$ seconds and a query $\mathcal{Q}$, \textbf{One-to-Many Video Temporal Grounding (OMTG)} aims to localize all $K \ge 1$ disjoint intervals:
\begin{equation}
    \mathcal{Y}^* = \big\{ [s_k^*, e_k^*] \big\}_{k=1}^{K}, \quad \text{with } 0 \le s_1^* < e_1^* < \dots < e_K^* \le T,
    \label{eq:omtg_formulation}
\end{equation}
where each segment $[s_k^*, e_k^*]$ satisfies $\mathcal{Q}$. OMTG requires solving both \emph{cardinality perception} (predicting $K$) and \emph{boundary localization} (predicting $[s_k, e_k]$). Direct whole-video inference with frozen Multimodal LLMs (MLLMs) causes visual token dilution and background distractor hallucinations, resulting in cardinality collapse ($0.00\%$ Count Accuracy).

To resolve this without training, \textbf{Adaptive Scout-Guided Dense Evidence (Adaptive SGDE)} decouples OMTG into four stages (Fig.~\ref{fig:pipeline}): (1) 1.0~FPS scouting ($\S$\ref{sec:stage1}), (2) hysteresis energy mining ($\S$\ref{sec:stage2}), (3) scale-invariant window geometry ($\S$\ref{sec:stage3}), and (4) budgeted dense decoding ($\S$\ref{sec:stage4}).

    \caption{\textbf{Adaptive SGDE architecture.} Decoupling scouting and adaptive windowing eliminates background dead-zones and restores multi-event cardinality.}
    \label{fig:pipeline}
\end{figure}

\subsection{Stage 1: Coarse-Grained Temporal Timeline Scouting}
\label{sec:stage1}

\paragraph{Formulation.} Ingesting full-length videos with an 8B Vision Transformer (ViT) incurs $O(T)$ FLOPs. We instead deploy a frozen 1.04B scout model $\mathcal{M}_{\text{scout}}$ (e.g., Llama-Nemotron-Embed-VL-1B) at 1.0~FPS. We compute frame embeddings $\mathbf{v}(t) = \mathcal{M}_{\text{scout}}(f_t)$ and query embedding $\mathbf{q} = \mathcal{M}_{\text{scout}}(\mathcal{Q})$, yielding raw cosine similarity:
\begin{equation}
    S(t) = \frac{\langle \mathbf{v}(t), \mathbf{q} \rangle}{\|\mathbf{v}(t)\|_2 \|\mathbf{q}\|_2}, \quad t \in [1, T].
    \label{eq:cosine_sim}
\end{equation}

\paragraph{Video-Adaptive $Z$-Score Normalization.} Because baseline similarity shifts across video domains and lighting, static thresholds fail. We apply video-level $Z$-score normalization:
\begin{equation}
    Z(t) = \frac{S(t) - \mu_S}{\sigma_S + \epsilon}, \quad \mu_S = \frac{1}{T}\sum_{t=1}^T S(t), \quad \sigma_S = \sqrt{\frac{1}{T}\sum_{t=1}^T \big(S(t) - \mu_S\big)^2},
    \label{eq:z_score}
\end{equation}
where $\epsilon = 10^{-6}$. Timeline $Z(t)$ isolates relative semantic surges and consumes $<2.5\%$ of total inference FLOPs.

\subsection{Stage 2: Dual-Threshold Hysteresis and Excess Energy Mining}
\label{sec:stage2}

\paragraph{Hysteresis Proposal Search.} Single thresholds either fragment long events (if too high) or trigger background noise (if too low). We employ dual-threshold hysteresis with trigger $\tau_{\text{high}} = 0.80$ and expansion $\tau_{\text{low}} = 0.25$. An interval $[a, b]$ is extracted if it contains a peak $Z(t_p) \ge \tau_{\text{high}}$ ($t_p \in [a, b]$) and expands outward while $Z(t) \ge \tau_{\text{low}}$, bounded by $Z(a-1) < \tau_{\text{low}}$ and $Z(b+1) < \tau_{\text{low}}$.

\paragraph{Excess Energy Integral.} To rank proposals without penalizing duration, we integrate excess semantic mass:
\begin{equation}
    J(a, b) = \sum_{t=a}^{b} \big(Z(t) - \tau_{\text{base}} - \lambda_{\text{len}}\big) \cdot \Delta t,
    \label{eq:excess_energy}
\end{equation}
where $\tau_{\text{base}} = 0.30$, $\lambda_{\text{len}} = 0.01$, and $\Delta t = 1.0\text{s}$. Each candidate $c = [a, b]$ is scored as $\text{Score}(c) = \max_{t \in [a, b]} Z(t) + 0.5 \max(0, J(a, b))$, followed by 1D NMS ($\text{IoU} \ge 0.30$). This improves ground-truth coverage from $70.9\%$ to $80.3\%$ and cuts missed action spans by $31.1\%$.

\subsection{Stage 3: Scale-Invariant Adaptive Window Geometry}
\label{sec:stage3}

\paragraph{Sub-Linear Margin and Gap.} Fixed temporal margins over-expand short clips and under-buffer long videos. We introduce duration-dependent sub-linear functions:
\begin{align}
    \text{Logarithmic Margin: } \Delta(T) &= \text{clamp}(3.5 \ln T, \, 8.0, \, 22.0)\text{ seconds}, \label{eq:log_margin} \\
    \text{Square-Root Gap: } G(T) &= \max(15.0, \, 3.5\sqrt{T})\text{ seconds}. \label{eq:sqrt_gap}
\end{align}
Proposals $c_i, c_{i+1}$ separated by distance $\le G(T)$ are bridged into a candidate cluster.

\paragraph{Soft Continuity Merge.} For candidate envelope $W_{\text{start}} = \min_i (a_i - \Delta(T))$ and $W_{\text{end}} = \max_i (b_i + \Delta(T))$, we merge intervals into a single corridor if the span ratio $\frac{W_{\text{end}} - W_{\text{start}}}{T} \ge \rho = 0.70$ or adjacent gap $\le G(T)$. This retains $95.0\%$ of videos in a single continuous prompt, preserving multi-event relational context while pruning $37.2\%-85.2\%$ of background dead-zones.

\subsection{Stage 4: Dense Grounding and Structured Multi-Span Decoding}
\label{sec:stage4}

\paragraph{Budgeted Corridor Allocation.} Rather than uniform video sampling, we allocate a fixed budget of $B = 128$ frames inside $[W_{\text{start}}, W_{\text{end}}]$. The local frame rate scales to $\text{FPS}_{\text{local}} = B / (W_{\text{end}} - W_{\text{start}}) \approx 4.0-6.0\text{ FPS}$ ($2\times-3\times$ denser than native 2.0~FPS), while cutting 8B ViT compute by $60.3\%$.

\paragraph{Structured JSON Decoding.} To prevent conversational format collapse, we use a schema-delimited prompt:
\begin{center}
\fbox{\parbox{0.92\textwidth}{
\small
\texttt{Given query: "\{query\}", return ALL relevant time spans in seconds.}\\
\texttt{Format: JSON array of [start, end] pairs: [[start1, end1], ...].}
}}
\end{center}
The primary LVLM $\mathcal{M}_{\text{ground}}$ (TimeLens-8B) decodes $\mathcal{Y} = \{[s_k, e_k]\}_{k=1}^K$ over the focused frames. For larger budgets ($B=512$), optical flow patch pruning evicts static background tokens. On OMTG Bench, Adaptive SGDE elevates Count Accuracy from $0.00\%$ to $33.12\%$ and Effective Temporal F1 from $0.00\%$ to $19.86\%$.
